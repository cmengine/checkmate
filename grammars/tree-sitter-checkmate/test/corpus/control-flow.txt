int classify(int score) {
    bool low = score < 10
    bool ready = (score > 50 && score < 90) || score == 100

    if (low) {
        return 0
    } else if (score >= 90) {
        return 2
    } else if (ready) {
        return 1
    } else {
        return -1
    }
}

void noElse(int n) {
    if (n == 0) {
        return
    }
    infer after = true
}

================================================================================
Newline-separated `else` re-attaches to the if (the real parser
backtracks over the break; GLR keeps both readings and the else wins)
================================================================================

int pick(int x) {
    if (x > 0) {
        return 1
    }
    else {
        return -1
    }
}
